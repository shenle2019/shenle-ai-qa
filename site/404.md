<script setup>
// 两种最常见的「地址手滑」自动纠一次，纠不回来就把地址显示出来
if (typeof window !== 'undefined') {
  const { pathname, origin } = window.location
  const bare = pathname.replace(/\/+$/, '')
  const inSite = bare.startsWith('/shenle-ai-qa')

  if (pathname !== bare && inSite) {
    window.location.replace(origin + bare)
  } else if (inSite && !pathname.endsWith('/')) {
    // 结尾少了斜杠的目录地址（/chapters → /chapters/）也纠一次
    fetch(origin + pathname + '/', { method: 'HEAD' })
      .then((r) => { if (r.ok) window.location.replace(origin + pathname + '/') })
      .catch(() => {})
  } else {
    const el = document.getElementById('nf-desc')
    if (el) el.textContent = '你要找的地址是：' + pathname
  }
}
</script>
